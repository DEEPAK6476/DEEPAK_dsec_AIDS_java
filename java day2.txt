
import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        int c = sc.nextInt();
        int largest = (a >= b && a >= c) ? a :
            (b >= a && b >= c) ? b : c;
        System.out.println(largest);
    }
}
